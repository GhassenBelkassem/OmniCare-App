class MedicalDocumentsScreen extends StatefulWidget {
  const MedicalDocumentsScreen({
    super.key,
    this.documentsApi,
    this.accountApi,
    this.initialPatientId,
    this.pickDocumentFile,
    this.pollDelay = const Duration(seconds: 2),
    this.maximumPolls = 6,
    this.maximumPollDelay = const Duration(minutes: 1),
    this.maximumAutomaticWatchDuration = const Duration(minutes: 15),
    this.retryJitterRatio = 0.20,
    this.randomDouble,
  }) : assert(retryJitterRatio >= 0 && retryJitterRatio <= 0.50);

  final DocumentsApi? documentsApi;
  final AccountApi? accountApi;

  // Add this field.
  final String? initialPatientId;

  final Future<PlatformFile?> Function()? pickDocumentFile;
  final Duration pollDelay;
  final int maximumPolls;
  final Duration maximumPollDelay;
  final Duration maximumAutomaticWatchDuration;
  final double retryJitterRatio;
  final double Function()? randomDouble;

  @override
  State<MedicalDocumentsScreen> createState() => _MedicalDocumentsScreenState();
}

class _MedicalDocumentsScreenState extends State<MedicalDocumentsScreen> {
  bool _isLoading = false;
  bool _pollInFlight = false;
  late final DocumentsApi _documentsApi = widget.documentsApi ?? DocumentsApi();
  late final AccountApi _accountApi = widget.accountApi ?? AccountApi();
  String? _patientId;
  List<MedicalDocumentRecord> _documents = const [];
  Timer? _pollTimer;
  int _pollCount = 0;
  DocumentUploadOperation? _pendingUpload;
  final Map<String, DocumentDeleteOperation> _pendingDeletes = {};

  @override
  void initState() {
    super.initState();
    _loadDocuments();
  }

  @override
  void dispose() {
    _pollTimer?.cancel();
    super.dispose();
  }

  String _toBackendDocumentType(String uiType) {
    final v = uiType.trim().toLowerCase();
    if (v == 'prescription') return 'PRESCRIPTION_DOCUMENT';
    if (v == 'lab results') return 'LAB_RESULT';
    if (v == 'scan') return 'IMAGING';
    return 'OTHER_MEDICAL_DOCUMENT';
  }

  String _mimeFor(String filename) {
    final lower = filename.toLowerCase();
    if (lower.endsWith('.pdf')) return 'application/pdf';
    if (lower.endsWith('.png')) return 'image/png';
    return 'image/jpeg';
  }

  Future<PlatformFile?> _pickDocumentFile() async {
    final injectedPicker = widget.pickDocumentFile;
    if (injectedPicker != null) return injectedPicker();
    final result = await FilePicker.platform.pickFiles(
      type: FileType.custom,
      allowedExtensions: ['pdf', 'jpg', 'jpeg', 'png'],
      withData: true,
    );
    return result?.files.single;
  }

  Future<void> _loadDocuments({bool showErrors = true}) async {
    if (_isLoading) return;
    setState(() => _isLoading = true);
    try {
      final profile = await _accountApi.getPatientProfile();
      final patientId = profile.patientId;
      if (patientId == null || patientId.isEmpty) {
        throw const ApiException('Your patient profile is not ready yet.');
      }

      // Upload availability must not depend on the initial document-list call.
      // Keep the valid profile patient ID even if MRS listing is temporarily down.
      if (!mounted) return;
      setState(() => _patientId = patientId);

      final page = await _documentsApi.listDocuments(patientId);
      if (!mounted) return;
      setState(() => _documents = page.documents);
      _startPollingIfNeeded();
    } catch (error) {
      if (showErrors) _showError(error);
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  Future<MedicalDocumentRecord> _uploadDocument({
    required String documentName,
    required String documentType,
    required PlatformFile file,
  }) async {
    if (_isLoading) {
      throw const ApiException('A document operation is already in progress.');
    }
    final patientId = _patientId;
    if (patientId == null) {
      throw const ApiException('Your patient profile is not ready yet.');
    }
    final bytes = file.bytes;
    if (bytes == null) {
      throw const ApiException('The selected file could not be read.');
    }
    setState(() => _isLoading = true);
    try {
      _pendingUpload ??= _documentsApi.startUpload(
        patientId: patientId,
        bytes: bytes,
        filename: file.name,
        mimeType: _mimeFor(file.name),
        displayName: documentName,
        documentType: _toBackendDocumentType(documentType),
        documentDate: DateTime.now(),
      );
      final uploaded = await _documentsApi.upload(_pendingUpload!);
      _pendingUpload = null;
      if (!mounted) return uploaded;
      setState(
        () => _documents = [
          uploaded,
          ..._documents.where((item) => item.id != uploaded.id),
        ],
      );
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text('Document status: ${uploaded.status}'),
          backgroundColor: const Color(0xFF1FBF9A),
        ),
      );
      if (!uploaded.isTerminal) _startPollingIfNeeded();
      return uploaded;
    } catch (error) {
      _showError(error);
      rethrow;
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  void _startPollingIfNeeded() {
    _pollTimer?.cancel();
    if (!_documents.any((document) => !document.isTerminal)) return;
    _pollCount = 0;
    _pollTimer = Timer(widget.pollDelay, _pollPendingDocuments);
  }

  Future<void> _pollPendingDocuments() async {
    if (!mounted || _pollInFlight || _pollCount >= widget.maximumPolls) return;
    final patientId = _patientId;
    if (patientId == null) return;
    _pollInFlight = true;
    _pollCount++;
    try {
      final pending = _documents
          .where((document) => !document.isTerminal)
          .toList();
      for (final document in pending) {
        final updated = await _documentsApi.getStatus(
          patientId: patientId,
          documentId: document.id,
        );
        if (!mounted) return;
        setState(
          () => _documents = _documents
              .map((item) => item.id == updated.id ? updated : item)
              .toList(),
        );
      }
      final hasPendingDocuments = _documents.any(
        (document) => !document.isTerminal,
      );
      if (hasPendingDocuments && _pollCount < widget.maximumPolls) {
        _pollTimer = Timer(widget.pollDelay, _pollPendingDocuments);
      } else if (hasPendingDocuments) {
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text(
                'Document processing is taking longer than expected. Refresh to check again.',
              ),
              backgroundColor: Color(0xFF667085),
            ),
          );
        }
      } else {
        await _loadDocuments(showErrors: false);
      }
    } catch (error) {
      _showError(error);
      if (mounted && _pollCount < widget.maximumPolls) {
        _pollTimer = Timer(widget.pollDelay, _pollPendingDocuments);
      }
    } finally {
      _pollInFlight = false;
    }
  }

  Future<void> _deleteDocument(MedicalDocumentRecord document) async {
    final patientId = _patientId;
    if (patientId == null || _isLoading) return;
    setState(() => _isLoading = true);
    try {
      final operation = _pendingDeletes.putIfAbsent(
        document.id,
        () => _documentsApi.startDelete(
          patientId: patientId,
          documentId: document.id,
        ),
      );
      await _documentsApi.delete(operation);
      _pendingDeletes.remove(document.id);
      if (mounted) setState(() => _isLoading = false);
      await _loadDocuments(showErrors: false);
    } catch (error) {
      _showError(error);
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  void _showError(Object error) {
    if (!mounted) return;
    final message = error is ApiException
        ? error.message
        : 'The document operation could not be completed. Please try again.';
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(message),
        backgroundColor: const Color(0xFFDC2626),
      ),
    );
  }
